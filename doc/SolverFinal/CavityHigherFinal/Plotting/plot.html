0063     % Draws element outlines for a structured quad mesh (works for both
0064     % Q4 geometry elements and higher-order solution elements). Node
0065     % ordering within an element isn't assumed -- the outline is drawn
0066     % via the convex hull of each element's nodes, which gives the
0067     % correct quad boundary for both straight-edged and higher-order
0068     % (curved-looking) elements on a structured mesh.
0069     hold on;
0070     numEl = size(conn,1);
0071     for e = 1:numEl
0072         nodes = conn(e,:);
0073         xe = x(nodes); ye = y(nodes);
0074         if numel(xe) >= 3
0075             k = convhull(xe(:), ye(:));
0076             patch(xe(k), ye(k), 'w', 'FaceColor','none', ...
0077                   'EdgeColor', color, 'LineWidth', 0.8);
0078         end
0079     end
0080     line(x, y, 'LineStyle','none', 'Marker','.', ...
0081          'Color', color*0.6, 'MarkerSize', 4);
